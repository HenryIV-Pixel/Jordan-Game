# 6. April 1957: the palace and the government cannot both survive.
jor.6 = {
	type = country_event
	title = jor.6.t
	desc = jor.6.d
	flavor = jor.6.f
	event_image = { video = "unspecific_politicians_arguing" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_protest.dds"
	duration = 3

	option = { # dismiss the government, ban the parties, martial law (history)
		name = jor.6.a
		default_option = yes
		set_variable = { name = jor_nabulsi_done }
		set_variable = { name = jor_palace_won }
		add_modifier = { name = jor_martial_law years = 3 }
		if = {
			limit = { has_modifier = jor_arab_subsidy }
			remove_modifier = jor_arab_subsidy
		}
		if = {
			limit = { NOT = { has_modifier = jor_british_subsidy } }
			add_modifier = { name = jor_us_aid years = 10 }
		}
		jor_relations = { TAG = USA VALUE = 40 }
		jor_relations = { TAG = EGY VALUE = -50 }
		jor_relations = { TAG = SYR VALUE = -40 }
		jor_relations = { TAG = SAU VALUE = 20 }
		jor_please = { IG = ig_armed_forces }
		jor_please = { IG = ig_landowners }
		jor_please = { IG = ig_devout }
		jor_crush = { IG = ig_intelligentsia }
		jor_crush = { IG = ig_trade_unions }
		ai_chance = { base = 70 }
	}
	option = { # let the elected government stand
		name = jor.6.b
		set_variable = { name = jor_nabulsi_done }
		add_modifier = { name = jor_officers_uneasy years = 2 }
		jor_relations = { TAG = EGY VALUE = 30 }
		jor_relations = { TAG = USA VALUE = -20 }
		jor_please = { IG = ig_intelligentsia }
